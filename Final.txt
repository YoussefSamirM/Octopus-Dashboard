        function csvToArray(strData) {
            var p = new RegExp(("(\\,|\\r?\\n|\\r|^)(?:\"([^\"]*(?:\"\"[^\"]*)*)\"|([^\"\\,\\r\\n]*))"), "gi");
            var a = [[]], m = null;
            while (m = p.exec(strData)) {
                if (m[1].length && m[1] !== ",") a.push([]);
                var v = m[2] ? m[2].replace(new RegExp("\"\"", "g"), "\"") : m[3];
                a[a.length - 1].push(v);
            }
            if (a.length > 0 && a[a.length-1].length === 1 && a[a.length-1][0] === "") a.pop();
            return a;
        }

        document.getElementById('inputStatus').addEventListener('change', function(e) {
            if (!e.target.files.length) return;

            btnText.innerText = "Parsing...";
            let reader = new FileReader();
            reader.onload = function(evt) {
                try {
                    let data = csvToArray(evt.target.result);
                    if (data.length < 2) { displayError("Status CSV is empty."); return; }
                    
                    let headers = data[0].map(h => h ? String(h).trim().toLowerCase() : '');
                    let eIdx = headers.findIndex(h => h === 'agent_email' || h === 'email' || h.includes('email') || h === 'sf user');
                    let sIdx = headers.findIndex(h => h === 'status');
                    let startIdx = headers.findIndex(h => h === 'date' || h === 'start' || h.includes('start'));
                    let endIdx = headers.findIndex(h => h === 'end_timestamp' || h === 'end' || h.includes('end'));

                    if (eIdx === -1 || startIdx === -1 || sIdx === -1) { displayError("Missing Columns in CSV."); return; }

                    let tempLogs = [];
                    for(let i=1; i<data.length; i++) {
                        let row = data[i];
                        if(!row || row.length <= eIdx || !row[eIdx]) continue;
                        let status = row[sIdx] ? String(row[sIdx]).toUpperCase().trim() : '';
                        let st = row[startIdx] ? new Date(row[startIdx].replace(/"/g, '')).getTime() : null;
                        let en = row[endIdx] && row[endIdx].trim() !== '' ? new Date(row[endIdx].replace(/"/g, '')).getTime() : new Date().getTime();
                        
                        if (st && en && !isNaN(st) && !isNaN(en) && status.includes('ONLINE') && en > st) {
                            let cleanEmail = row[eIdx].toLowerCase().trim().split('@')[0].replace(/[^a-z0-9]/g, '');
                            tempLogs.push({ email: cleanEmail, start: st, end: en, name: cleanEmail });
                        }
                    }

                    let agentGroups = {};
                    tempLogs.forEach(log => {
                        if(!agentGroups[log.email]) agentGroups[log.email] = [];
                        agentGroups[log.email].push(log);
                    });

                    window.loadedFiles.statusParsed = [];
                    Object.keys(agentGroups).forEach(email => {
                        let logs = agentGroups[email].sort((a, b) => a.start - b.start);
                        let merged = [logs[0]];
                        for (let i = 1; i < logs.length; i++) {
                            let curr = logs[i];
                            let last = merged[merged.length - 1];
                            if (curr.start <= last.end) { last.end = Math.max(last.end, curr.end); } 
                            else { merged.push(curr); }
                        }
                        merged.forEach(m => window.loadedFiles.statusParsed.push(m));
                    });

                    document.getElementById('btnStatus').parentElement.querySelector('.btn-upload').classList.add('success'); 
                    btnText.innerText = "Log Loaded (" + window.loadedFiles.statusParsed.length + " rows)";
                } catch(err) { displayError("Error reading Status CSV."); }
            };
            reader.readAsText(file, "UTF-8");
        });

        function formatTimeSecs(totalSeconds) {
            if (totalSeconds === undefined || totalSeconds === null || totalSeconds <= 0) return "0:00:00";
            totalSeconds = Math.round(totalSeconds);
            let h = Math.floor(totalSeconds / 3600);
            let m = Math.floor((totalSeconds % 3600) / 60);
            let s = totalSeconds % 60;
            return `${h}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
        }

        function formatTimeFromMs(ms) {
            if (ms <= 0) return "00:00";
            let totalSeconds = Math.round(ms / 1000);
            let m = Math.floor(totalSeconds / 60);
            let s = totalSeconds % 60;
            return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
        }

        function formatPerc(val) { 
            return (val || 0).toFixed(2) + "%"; 
        }

        function getDatesInRange(sDateStr, eDateStr) {
            let dates = [];
            let curr = new Date(sDateStr + "T00:00:00");
            let endStr = eDateStr ? eDateStr : sDateStr;
            let end = new Date(endStr + "T00:00:00");
            end.setDate(end.getDate() + 1); 
            while(curr <= end) {
                let y = curr.getFullYear(), m = String(curr.getMonth()+1).padStart(2,'0'), d = String(curr.getDate()).padStart(2,'0');
                dates.push(`${y}-${m}-${d}`);
                curr.setDate(curr.getDate() + 1);
            }
            return dates;
        }

        function groupLOB(raw) {
            if (!raw) return "Unknown LOB";
            let l = String(raw).toLowerCase().trim().replace(/[\s\-\_\.]/g, '');
            if (l.includes('supportnonpilot') || l.includes('supportnonpolit') || l.includes('nonpilot') || l.includes('support')) return "Combined";
            if (l.includes('tmart') && (l.includes('followup') || l.includes('fu'))) return "TMart-FU";
            if (l.includes('tmart') || l.includes('fcr') || l.includes('delivery') || l.includes('pickup') || l.includes('dp')) return "Combined";
            if (l.includes('tpro')) return "TPro";
            if (l.includes('ghc')) return "GHC";
            return String(raw).trim(); 
        }

        function parseDurationSecs(val) {
            if (val === undefined || val === null || val === '') return 0;
            if (typeof val === 'number') return Math.round(val * 24 * 3600); 
            var pts = val.toString().replace(/[a-zA-Z\s]/g, '').split(':');
            var h = parseInt(pts[0], 10) || 0;
            var m = pts.length > 1 ? parseInt(pts[1], 10) : 0;
            var s = pts.length > 2 ? parseInt(pts[2], 10) : 0;
            return (h * 3600) + (m * 60) + s;
        }

        function parseInterval(str) {
            let s = String(str).toLowerCase().trim();
            let isPM = s.includes('pm'), isAM = s.includes('am');
            let tm = s.replace(/[^0-9:]/g, '').split(':');
            if (tm.length > 0 && tm[0] !== '') {
                let th = parseInt(tm[0], 10), tmins = tm.length > 1 ? parseInt(tm[1], 10) : 0;
                if (isPM && th !== 12) th += 12; if (isAM && th === 12) th = 0;
                return th * 60 + tmins;
            }
            return -1;
        }

        function executeParser(reqDates) {
            let dateTargets = reqDates.map(function(dStr) {
                var dObj = new Date(dStr + "T00:00:00");
                var mNames = ["jan","feb","mar","apr","may","jun","jul","aug","sep","oct","nov","dec"];
                return {
                    iso: dStr,
                    strMatch: new RegExp("(^|\\D)0?" + dObj.getDate() + "(?:st|nd|rd|th)?\\-?" + mNames[dObj.getMonth()], "i"),
                    y: dObj.getFullYear(), m: dObj.getMonth(), d: dObj.getDate(),
                    monthStr: mNames[dObj.getMonth()]
                };
            });

            // 1. Process Skills Matrix
            window.agentInfo = {};
            let skillsWb = window.loadedFiles.skillsWorkbook;
            let sSheetName = skillsWb.SheetNames[0];
            let sDataDisp = XLSX.utils.sheet_to_json(skillsWb.Sheets[sSheetName], {header: 1, raw: false, defval: ""});
            
            let sfUserIdx = -1, hrIdx = -1, nameIdx = -1, tlIdx = -1, osvIdx = -1;
            let headerRowIdx = -1;
            let dateColMapSkills = {};

            for(let r=0; r<Math.min(20, sDataDisp.length); r++) {
                if(!sDataDisp[r]) continue;
                for(let c=0; c<sDataDisp[r].length; c++) {
                    let cell = String(sDataDisp[r][c]).toLowerCase().trim();
                    if(!cell) continue;

                    if(sfUserIdx === -1 && (cell === 'sf user' || cell.includes('email') || cell === 'agent_email')) {
                        sfUserIdx = c; headerRowIdx = r;
                    }
                    if(cell === 'hr id' || cell === 'hr') hrIdx = c;
                    if(cell === 'agent name' || cell === 'name' || cell === 'agent_name') nameIdx = c;
                    if(cell === 'tl' || cell === 'team leader') tlIdx = c;
                    if(cell === 'osv') osvIdx = c;

                    dateTargets.forEach(dt => {
                        if(dateColMapSkills[dt.iso] === undefined) {
                            if(dt.strMatch.test(cell)) {
                                dateColMapSkills[dt.iso] = c;
                            }
                        }
                    });
                }
                if(sfUserIdx !== -1) break;
            }

            if(sfUserIdx !== -1) {
                for(let row = headerRowIdx + 1; row < sDataDisp.length; row++) {
                    let email = sDataDisp[row][sfUserIdx];
                    if(email) {
                        let cleanEmail = email.toString().toLowerCase().trim().split('@')[0].replace(/[^a-z0-9]/g, '');
                        window.agentInfo[cleanEmail] = {
                            hr: hrIdx !== -1 ? sDataDisp[row][hrIdx] : '-',
                            name: nameIdx !== -1 ? sDataDisp[row][nameIdx] : cleanEmail,
                            tl: tlIdx !== -1 ? sDataDisp[row][tlIdx] : '-',
                            osv: osvIdx !== -1 ? sDataDisp[row][osvIdx] : '-',
                            lobs: {}
                        };
                        
                        dateTargets.forEach(dt => {
                            let cIdx = dateColMapSkills[dt.iso];
                            if(cIdx !== undefined) {
                                let lob = sDataDisp[row][cIdx];
                                if(lob) {
                                    let cleanLob = lob.toString().trim();
                                    if(cleanLob && cleanLob !== '-') {
                                        window.agentInfo[cleanEmail].lobs[dt.iso] = groupLOB(cleanLob);
                                    }
                                }
                            }
                        });
                    }
                }
            }

            // 2. Process ABS & HC Data exactly as per image
            let absWb = window.loadedFiles.absWorkbook;
            let dailyAbsData = {}; 
            let absSheet = absWb.Sheets[absWb.SheetNames[0]]; 
            let absDataDisp = XLSX.utils.sheet_to_json(absSheet, {header: 1, raw: false, defval: ""});
            let absDataRaw = XLSX.utils.sheet_to_json(absSheet, {header: 1, raw: true, defval: ""});

            let dateColsAbs = {}; 
            for(let r=0; r<Math.min(10, absDataDisp.length); r++) {
                let foundDate = false;
                for(let c=0; c<absDataDisp[r].length; c++) {
                    let cell = String(absDataDisp[r][c]).trim();
                    dateTargets.forEach(dt => {
                        if(dt.strMatch.test(cell) || cell.includes(dt.d + "-" + dt.monthStr)) {
                            dateColsAbs[c] = dt.iso;
                            foundDate = true;
                        }
                    });
                }
                if(foundDate) break; 
            }

            Object.keys(dateColsAbs).forEach(c => {
                let iso = dateColsAbs[c];
                dailyAbsData[iso] = {};
                for(let r=0; r<absDataDisp.length; r++) {
                    let val = String(absDataDisp[r][c]).toLowerCase().replace(/[\s\-\_\.]/g, '');
                    let matchedLob = null;
                    if(val === 'combined') matchedLob = 'Combined';
                    else if(val === 'tpro' || val === 't-pro') matchedLob = 'TPro';
                    else if(val === 'ghc') matchedLob = 'GHC';
                    else if(val.includes('tmart') && (val.includes('fu') || val.includes('followup'))) matchedLob = 'TMart-FU';

                    if(matchedLob) {
                        let hcRaw = absDataRaw[r+1] ? absDataRaw[r+1][c] : 0;
                        let absRaw = absDataRaw[r+3] ? absDataRaw[r+3][c] : 0;
                        let absPercRaw = absDataRaw[r+4] ? absDataRaw[r+4][c] : 0;

                        let absPercVal = 0;
                        if (typeof absPercRaw === 'number') absPercVal = absPercRaw * 100;
                        else absPercVal = parseFloat(String(absPercRaw).replace('%', ''));

                        dailyAbsData[iso][matchedLob] = {
                            hc: parseDurationSecs(hcRaw), 
                            abs: parseDurationSecs(absRaw),
                            absPerc: isNaN(absPercVal) ? 0 : absPercVal
                        };
                    }
                }
            });

            // 3. Initialize Master Data storage
            let dailyRawData = {}; 
            sortedDates = [];
            for(let i=0; i<reqDates.length - 1; i++) {
                let dIso = reqDates[i];
                sortedDates.push(dIso);
                dailyRawData[dIso] = {};
                LOBs.forEach(lob => {
                    dailyRawData[dIso][lob.id] = {};
                    for(let k=0; k<=1410; k+=30) {
                        dailyRawData[dIso][lob.id][k] = { req:0, act:0, bill:0, over:0, lost:0 };
                    }
                });
            }

            let masterWb = window.loadedFiles.masterWorkbook;
            masterWb.SheetNames.forEach(sNameOriginal => {
                var sInfo = masterWb.Workbook && masterWb.Workbook.Sheets ? masterWb.Workbook.Sheets.find(s => s.name === sNameOriginal) : null;
                if(sInfo && sInfo.Hidden !== 0) return;

                let sName = sNameOriginal.toLowerCase().trim();
                let matchedDt = dateTargets.find(dt => dt.strMatch.test(sName));
                if(!matchedDt || !dailyRawData[matchedDt.iso]) return;
                
                let dIso = matchedDt.iso;
                let sheetData = XLSX.utils.sheet_to_json(masterWb.Sheets[sNameOriginal], {header: 1, raw: false, defval: ""});

                let lobCols = {};
                for(let r=0; r<Math.min(10, sheetData.length); r++) {
                    if(!sheetData[r]) continue;
                    for(let c=0; c<sheetData[r].length; c++) {
                        let val = String(sheetData[r][c]).toLowerCase().replace(/[\s\-\_\.]/g, '');
                        let matchedLob = null;
                        if(val === 'combined') matchedLob = 'Combined';
                        else if(val === 'tpro' || val === 't-pro') matchedLob = 'TPro';
                        else if(val === 'ghc') matchedLob = 'GHC';
                        else if(val.includes('tmart') && (val.includes('fu') || val.includes('followup'))) matchedLob = 'TMart-FU';

                        if(matchedLob && !lobCols[matchedLob]) {
                            let intCol = -1;
                            let dataRow = -1;
                            for(let r2=r; r2<Math.min(r+6, sheetData.length); r2++) {
                                if(!sheetData[r2]) continue;
                                for(let c2=Math.max(0, c-4); c2<=c+4; c2++) {
                                    if(String(sheetData[r2][c2]).toLowerCase().includes('interval')) {
                                        intCol = c2;
                                        dataRow = r2 + 1;
                                        break;
                                    }
                                }
                                if(intCol !== -1) break;
                            }
                            if(intCol !== -1) {
                                lobCols[matchedLob] = { r: dataRow, c: intCol };
                            }
                        }
                    }
                }

                Object.keys(lobCols).forEach(lobId => {
                    let startRow = lobCols[lobId].r; 
                    for(let r=startRow; r<sheetData.length; r++) {
                        if(!sheetData[r]) continue;
                        let intStr = String(sheetData[r][lobCols[lobId].c]).trim();
                        if(!intStr || intStr.toLowerCase().includes('total')) continue;
                        
                        let sk = parseInterval(intStr);
                        if(sk !== -1 && dailyRawData[dIso][lobId][sk]) {
                            let req = parseDurationSecs(sheetData[r][lobCols[lobId].c + 1]);
                            let act = parseDurationSecs(sheetData[r][lobCols[lobId].c + 2]);
                            let bill = parseDurationSecs(sheetData[r][lobCols[lobId].c + 3]);
                            let over = parseDurationSecs(sheetData[r][lobCols[lobId].c + 4]);
                            let lost = parseDurationSecs(sheetData[r][lobCols[lobId].c + 5]);

                            dailyRawData[dIso][lobId][sk] = { req: req, act: act, bill: bill, over: over, lost: lost };
                        }
                    }
                });
            });

            // 4. Structure Data for STD / OVN
            globalProcessedData = { lobs: {} };
            LOBs.forEach(l => globalProcessedData.lobs[l.id] = { dates: {} });

            sortedDates.forEach(dIso => {
                let dtNext = new Date(dIso + "T00:00:00"); dtNext.setDate(dtNext.getDate() + 1);
                let nxStr = dtNext.toISOString().split('T')[0];

                LOBs.forEach(lobConf => {
                    let lob = lobConf.id;
                    let absStats = dailyAbsData[dIso] && dailyAbsData[dIso][lob] ? dailyAbsData[dIso][lob] : {hc:0, abs:0, absPerc:0};
                    
                    let stdInts = [], ovnInts = [];
                    for(let k=0; k<=1410; k+=30) {
                        let obj = dailyRawData[dIso][lob][k] || {req:0, act:0, bill:0, over:0, lost:0};
                        stdInts.push({ sk: k, iso: dIso, req: obj.req, act: obj.act, bill: obj.bill, over: obj.over, lost: obj.lost });
                    }
                    for(let k=480; k<=1410; k+=30) {
                        let obj = dailyRawData[dIso][lob][k] || {req:0, act:0, bill:0, over:0, lost:0};
                        ovnInts.push({ sk: k, iso: dIso, req: obj.req, act: obj.act, bill: obj.bill, over: obj.over, lost: obj.lost });
                    }
                    if(dailyRawData[nxStr] && dailyRawData[nxStr][lob]) {
                        for(let k=0; k<=450; k+=30) {
                            let obj = dailyRawData[nxStr][lob][k] || {req:0, act:0, bill:0, over:0, lost:0};
                            ovnInts.push({ sk: k, iso: nxStr, req: obj.req, act: obj.act, bill: obj.bill, over: obj.over, lost: obj.lost });
                        }
                    }

                    globalProcessedData.lobs[lob].dates[dIso] = {
                        std: aggregateShift(stdInts, absStats),
                        ovn: aggregateShift(ovnInts, absStats)
                    };
                });
            });
        }

        function aggregateShift(intervals, absStats) {
            let dayData = { req: 0, act: 0, bill: 0, over: 0, lost: 0, hc: absStats.hc, abs: absStats.abs, absPerc: absStats.absPerc, intervals: intervals };
            intervals.forEach(intObj => {
                dayData.req += intObj.req;
                dayData.act += intObj.act;
                dayData.bill += intObj.bill;
            });

            // STRICT NET OVERAGE / SHORTAGE logic
            if(dayData.act > dayData.req) {
                dayData.over = dayData.act - dayData.req;
                dayData.lost = 0;
            } else {
                dayData.lost = dayData.req - dayData.act;
                dayData.over = 0;
            }
            return dayData;
        }

        function renderLobDaysView() {
            let lobId = navState.lob;
            let lobConf = LOBs.find(l=>l.id===lobId);
            document.getElementById('lobDaysTitle').innerText = lobConf.title + " Daily Breakdown";
            
            let html = '';
            let tReq=0, tAct=0, tBill=0, tOver=0, tLost=0, tHc=0, tAbs=0;

            sortedDates.forEach(dIso => {
                let d = globalProcessedData.lobs[lobId].dates[dIso][currentShiftMode];
                if(d.req > 0 || d.act > 0 || d.hc > 0) {
                    tReq += d.req; tAct += d.act; tBill += d.bill; tOver += d.over; tLost += d.lost; tHc += d.hc; tAbs += d.abs;
                    
                    let schHrs = d.hc * 0.89; // Prod SCH
                    let schPerc = d.req > 0 ? (schHrs / d.req) * 100 : 0; // Updated SCH% formula
                    
                    let ic = d.req === 0 ? 100 : (d.bill / d.req) * 100;
                    let ovPerc = d.req > 0 ? (d.over / d.req) * 100 : 0;

                    html += `<tr>
                        <td class="val-bold">${dIso}</td>
                        <td class="val-bold" style="background:#f8fafc;">${formatTimeSecs(d.req)}</td>
                        <td class="val-bold">${formatTimeSecs(d.act)}</td>
                        <td class="val-danger">${formatPerc(d.absPerc)}</td>
                        <td class="val-danger">${formatTimeSecs(d.abs)}</td>
                        <td class="val-bold" style="color:var(--text-muted)">${formatTimeSecs(schHrs)}</td>
                        <td class="val-bold" style="color:var(--text-muted)">${formatTimeSecs(d.hc)}</td>
                        <td class="val-bold" style="color:var(--text-muted)">${formatPerc(schPerc)}</td>
                        <td class="val-accent">${formatTimeSecs(d.bill)}</td>
                        <td class="${ic >= 100 ? 'val-success' : 'val-danger'}">${formatPerc(ic)}</td>
                        <td class="val-danger">${formatTimeSecs(d.lost)}</td>
                        <td class="val-success">${formatTimeSecs(d.over)}</td>
                        <td class="val-success">${formatPerc(ovPerc)}</td>
                        <td><a class="action-link" onclick="navigateTo('interval', '${lobId}', '${dIso}')">Intervals <i class="fa-solid fa-arrow-right"></i></a></td>
                    </tr>`;
                }
            });

            let netOver = 0, netLost = 0;
            if(tAct > tReq) { netOver = tAct - tReq; netLost = 0; }
            else { netLost = tReq - tAct; netOver = 0; }

            let tSchHrs = tHc * 0.89;
            let tAbsPerc = tHc > 0 ? (tAbs / tHc) * 100 : 0;
            let tSchPerc = tReq > 0 ? (tSchHrs / tReq) * 100 : 0; // Updated SCH% formula
            let tOvPerc = tReq > 0 ? (netOver / tReq) * 100 : 0;
            let mIc = tReq === 0 ? 100 : (tBill / tReq) * 100;

            html += `<tr class="row-total">
                <td>Total</td>
                <td class="val-bold" style="background:#f8fafc;">${formatTimeSecs(tReq)}</td>
                <td class="val-bold">${formatTimeSecs(tAct)}</td>
                <td class="val-danger">${formatPerc(tAbsPerc)}</td>
                <td class="val-danger">${formatTimeSecs(tAbs)}</td>
                <td class="val-bold" style="color:var(--text-muted)">${formatTimeSecs(tSchHrs)}</td>
                <td class="val-bold" style="color:var(--text-muted)">${formatTimeSecs(tHc)}</td>
                <td class="val-bold" style="color:var(--text-muted)">${formatPerc(tSchPerc)}</td>
                <td class="val-accent">${formatTimeSecs(tBill)}</td>
                <td class="${mIc >= 100 ? 'val-success' : 'val-danger'}">${formatPerc(mIc)}</td>
                <td class="val-danger">${formatTimeSecs(netLost)}</td>
                <td class="val-success">${formatTimeSecs(netOver)}</td>
                <td class="val-success">${formatPerc(tOvPerc)}</td>
                <td>-</td>
            </tr>`;
            document.getElementById('lobDaysTableBody').innerHTML = html;
        }

        function renderAgentDetailsView() {
            let lobId = navState.lob;
            let dIso = navState.date;
            let sk = navState.sk;
            
            let hR = Math.floor(sk / 60), mR = sk % 60;
            let lbl = `${hR}:${mR === 0 ? '00' : '30'}`;
            document.getElementById('agentDetailTitle').innerText = `Details: ${LOBs.find(l=>l.id===lobId).title} (${dIso} @ ${lbl})`;

            let dayD = globalProcessedData.lobs[lobId].dates[dIso][currentShiftMode];
            let intObj = dayD.intervals.find(x => x.sk === sk);

            let intStartMs = new Date(dIso + "T00:00:00").getTime() + (sk * 60000);
            let intEndMs = intStartMs + 30 * 60000;
            
            let agentsOnline = [];
            
            window.loadedFiles.statusParsed.forEach(log => {
                if (log.end <= intStartMs || log.start >= intEndMs) return;
                
                let info = window.agentInfo[log.email] || { hr: '-', name: log.email, tl: '-', osv: '-', lobs: {} };
                let effLob = info.lobs[dIso] || 'Unknown LOB';
                
                if(effLob === lobId) {
                    let overlapStart = Math.max(log.start, intStartMs);
                    let overlapEnd = Math.min(log.end, intEndMs);
                    let durMs = overlapEnd - overlapStart;
                    
                    if(durMs > 0) {
                        let existing = agentsOnline.find(a => a.email === log.email);
                        if(existing) {
                            existing.durMs += durMs;
                        } else {
                            agentsOnline.push({
                                email: log.email, hr: info.hr, name: info.name, tl: info.tl, osv: info.osv, durMs: durMs
                            });
                        }
                    }
                }
            });

            agentsOnline.sort((a,b) => b.durMs - a.durMs);

            let tls = new Set();
            let osvs = new Set();
            agentsOnline.forEach(a => {
                if(a.tl && a.tl !== '-') tls.add(a.tl);
                if(a.osv && a.osv !== '-') osvs.add(a.osv);
            });
            
            let tlHtml = '<option value="">All Team Leaders</option>';
            Array.from(tls).sort().forEach(t => tlHtml += `<option value="${t}">${t}</option>`);
            document.getElementById('tlList').innerHTML = tlHtml;
            
            let osvHtml = '<option value="">All Supervisors</option>';
            Array.from(osvs).sort().forEach(o => osvHtml += `<option value="${o}">${o}</option>`);
            document.getElementById('osvList').innerHTML = osvHtml;

            let html = '';
            agentsOnline.forEach(a => {
                html += `<tr>
                    <td class="val-bold" style="color:var(--text-muted)">${a.hr}</td>
                    <td style="font-weight:700;">${a.name}</td>
                    <td>${a.email}</td>
                    <td>${a.tl}</td>
                    <td>${a.osv}</td>
                    <td class="val-bold val-success">${formatTimeFromMs(a.durMs)}</td>
                </tr>`;
            });
            document.getElementById('agentDetailTableBody').innerHTML = html !== '' ? html : '<tr><td colspan="6" style="text-align:center;">No Agents found online.</td></tr>';

            if(intObj) {
                let ic = intObj.req === 0 ? 100 : (intObj.act / intObj.req) * 100;
                document.getElementById('mmKpis').innerHTML = `
                    <div class="kpi-card" style="border-top-color:var(--tpro)"><div class="kpi-title">Interval REQ</div><div class="kpi-val">${formatTimeSecs(intObj.req)}</div></div>
                    <div class="kpi-card" style="border-top-color:var(--accent)"><div class="kpi-title">Interval Actual</div><div class="kpi-val">${formatTimeSecs(intObj.act)}</div></div>
                    <div class="kpi-card" style="border-top-color:var(--text-muted)"><div class="kpi-title">Agents Online</div><div class="kpi-val">${agentsOnline.length}</div></div>
                    <div class="kpi-card" style="border-top-color:var(--success)"><div class="kpi-title">Actual IC%</div><div class="kpi-val ${ic >= 100 ? 'val-success' : 'val-danger'}">${formatPerc(ic)}</div></div>
                `;
            }
        }
