function getDistroData() {

  try {

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();

    var data = sheet.getDataRange().getValues();

    var distroMap = {};

    if (data.length < 2) return JSON.stringify({ success: true, data: distroMap });

    var headers = data[0].map(function(h) { return h.toString().toLowerCase().trim(); });

    var emailIdx = headers.findIndex(function(h) { return h === 'talabat email' || h === 'email' || h.indexOf('email') > -1; });

    var svIdx = headers.findIndex(function(h) { return h.indexOf('2nd level') > -1 || h.indexOf('supervisor') > -1 || h === 'sv'; });

    var tlIdx = headers.findIndex(function(h) { return h.indexOf('1st level') > -1 || h.indexOf('tl') > -1 || h.indexOf('team leader') > -1; });

    var hrIdx = headers.findIndex(function(h) { return h === 'hr id' || h === 'hr'; });

    var lobIdx = headers.findIndex(function(h) { return h === 'lob' || h.indexOf('business') > -1; });

    var siteIdx = headers.findIndex(function(h) { return h === 'site' || h === 'location' || h.indexOf('site') > -1; });



    if (emailIdx === -1) return JSON.stringify({ success: false, error: "Missing Email Column" });



    for (var i = 1; i < data.length; i++) {

      var row = data[i];

      if (row[emailIdx]) {

        var rawEmail = row[emailIdx].toString().toLowerCase();

        var emailKey = rawEmail.replace('@talabat.com', '').replace('.ext', '').replace(/[\s\.]/g, '').trim();

        if (emailKey) {

          distroMap[emailKey] = {

            hr: hrIdx !== -1 && row[hrIdx] ? row[hrIdx].toString().trim() : 'N/A',

            sv: svIdx !== -1 && row[svIdx] ? row[svIdx].toString().trim() : 'Unassigned SPV',

            tl: tlIdx !== -1 && row[tlIdx] ? row[tlIdx].toString().trim() : 'Unassigned TL',

            lob: lobIdx !== -1 && row[lobIdx] ? row[lobIdx].toString().trim() : 'Unknown LOB',

            site: siteIdx !== -1 && row[siteIdx] ? row[siteIdx].toString().trim() : 'Unknown Site'

          };

        }

      }

    }

    return JSON.stringify({ success: true, data: distroMap });

  } catch (e) {

    return JSON.stringify({ success: false, error: e.toString() });

  }

}



function getHeatmapColor(val) {

  if (val === '-' || val === '' || isNaN(val) || val === 0) return '#ffffff';

 

  if (val < 360) return '#57bb8a';

  if (val < 390) return '#abc878';

  if (val < 420) return '#bfcb74';

  if (val < 450) return '#d2ce70';

  if (val < 480) return '#fbc768';

  if (val < 510) return '#f2a66d';

  return '#e67c73';                

}
